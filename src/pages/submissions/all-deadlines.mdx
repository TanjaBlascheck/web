import { getCollection } from 'astro:content';

export const allDeadlines = await getCollection('deadlines');

export const formatDate = (dateStr) => {
  return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
};

{/* Sort all deadlines chronologically by their final target date (extended or original) */}
export const sortedDeadlines = [...allDeadlines].sort((a, b) => {
  const dateA = new Date(a.data.extendedDate || a.data.originalDate);
  const dateB = new Date(b.data.extendedDate || b.data.originalDate);
  return dateA.getTime() - dateB.getTime();
});

{/* Group deadlines by category */}
export const groupedDeadlines = sortedDeadlines.reduce((groups, item) => {
  const category = item.data.category || 'Other';
  if (!groups[category]) groups[category] = [];
  groups[category].push(item);
  return groups;
}, {});

# Important Dates

{Object.entries(groupedDeadlines).map(([category, items]) => (
  <div key={category} className="category-section" style={{ marginBottom: '2rem' }}>
    
    {/* Small heading for each category */}
    <h3>{category}</h3>
    
    <table>
      <tbody>
        {items.map((item) => (
          <tr class="deadline-tracker" data-original={item.data.originalDate} data-extended={item.data.extendedDate || ""}>
            <td>{item.data.type}</td>
            <td>
              <span class="date-display">
                {item.data.extendedDate ? (
                  <>
                    <span class="crossed-out">{formatDate(item.data.originalDate)}</span>{" "}
                    <span class="extension">{formatDate(item.data.extendedDate)}</span>
                  </>
                ) : (
                  <span>{formatDate(item.data.originalDate)}</span>
                )}
              </span>
              <span class="live-badge"></span>
            </td>
          </tr>
        ))}
      </tbody>
    </table>
  </div>
))}
